# requirements-prod.txt
# ─────────────────────────────────────────────────────────────────────────────
# Minimal runtime dependencies for the DEPLOYED WEB SERVER ONLY.
#
# This is intentionally a small subset of requirements.txt — the full
# file also includes training-only packages (mlflow, optuna, wandb,
# stable-baselines3, seaborn, plotly, pytest, black) that the live API
# never imports. Installing those in production wastes build time,
# image size, and attack surface for no benefit.
#
# Use this file for:
#   - Docker production image (see Dockerfile)
#   - Render / Railway / any PaaS build step
#   - Any environment that only serves the API + runs PPO inference
#
# Use the full requirements.txt for:
#   - Local development
#   - Training (train_kt.py, train_ppo.py)
#   - Evaluation / analysis scripts (evaluate.py, ablation.py, learning_gain.py)
#
# Install with:
#     pip install -r requirements-prod.txt
# ─────────────────────────────────────────────────────────────────────────────


# ── ML inference (PPO policy forward pass only — no training) ────────────────

# Plain PyPI torch (CPU+CUDA bundled wheel). This is the safest default
# for portability across PaaS providers — some build environments don't
# reliably reach download.pytorch.org's custom package index, and the
# extra-index-url syntax for the slimmer +cpu-only wheel is finicky across
# pip versions and platforms. The bundled wheel is larger (~750MB vs
# ~190MB) but installs everywhere with zero extra configuration.
#
# If your platform's build step DOES reach download.pytorch.org and you
# want the smaller image, replace the line below with:
#     --extra-index-url https://download.pytorch.org/whl/cpu
#     torch==2.2.0+cpu
# and test the build once before relying on it in production.
torch==2.2.0

numpy==1.26.3
